─────────────────────────────────────────────────────────
  1️⃣  CEK SEMUA DATA FIREBASE (RINGKASAN)
─────────────────────────────────────────────────────────

firebase.database().ref('penilaian').once('value').then(snap => {
  const data = snap.val();
  console.log('\n📊 FIREBASE DATA SUMMARY');
  console.log('═'.repeat(50));
  console.log('Total inovasi:', Object.keys(data || {}).length);
  
  Object.entries(data || {}).forEach(([key, val]) => {
    console.log('\n🏆', key);
    console.log('  Juri Judul:', Object.keys(val.judulState || {}).join(', ') || '(belum ada)');
    console.log('  Juri Indikator:', Object.keys(val.juriState || {}).join(', ') || '(belum ada)');
    console.log('  Last update:', val.savedAt || '(tidak ada)');
  });
  console.log('\n' + '═'.repeat(50));
});

  2️⃣  CEK DATA INOVASI SPESIFIK + SKOR
─────────────────────────────────────────────────────────

// GANTI INI dengan nama inovasi (spasi → underscore):
const inovasiKey = "Ayo_Sekolah";  // ← GANTI DI SINI

firebase.database().ref('penilaian').child(inovasiKey).once('value').then(snap => {
  const data = snap.val();
  console.log('\n📋 DETAIL INOVASI:', inovasiKey.replace(/_/g, ' '));
  console.log('═'.repeat(60));
  
  if (!data) {
    console.log('❌ Data tidak ditemukan!');
    return;
  }
  
  console.log('\n✅ JURI JUDUL:');
  Object.entries(data.judulState || {}).forEach(([juri, state]) => {
    const skor = data.skorJudulPerJuri?.[juri] || 0;
    const itemCount = Object.keys(state).length;
    console.log(`  • ${juri}`);
    console.log(`    Skor: ${skor} | Item dinilai: ${itemCount}`);
  });
  
  console.log('\n✅ JURI INDIKATOR:');
  Object.entries(data.juriState || {}).forEach(([juri, state]) => {
    const skor = data.skorPerJuri?.[juri] || 0;
    const itemCount = Object.keys(state.radio || {}).length;
    console.log(`  • ${juri}`);
    console.log(`    Skor: ${skor} | Item dinilai: ${itemCount}`);
  });
  
  console.log('\n📅 Metadata:');
  console.log('  Perangkat Daerah:', data.perangkatDaerah || '(tidak ada)');
  console.log('  Kategori:', data.kategori || '(tidak ada)');
  console.log('  Last saved:', data.savedAt || '(tidak ada)');
  
  console.log('\n' + '═'.repeat(60));
});

  7️⃣  COMPARE: FIREBASE vs LOCALSTORAGE
─────────────────────────────────────────────────────────

Promise.all([
  firebase.database().ref('penilaian').once('value'),
  Promise.resolve(JSON.parse(localStorage.getItem('draf_iid2026_all') || '{}'))
]).then(([fbSnap, localData]) => {
  const fbData = fbSnap.val() || {};
  
  console.log('\n🔄 FIREBASE vs LOCALSTORAGE COMPARISON');
  console.log('═'.repeat(60));
  console.log('Firebase items:', Object.keys(fbData).length);
  console.log('LocalStorage items:', Object.keys(localData).length);
  
  // Inovasi hanya di Firebase
  const fbOnly = Object.keys(fbData).filter(k => !localData[k]);
  if (fbOnly.length > 0) {
    console.log('\n☁️  Hanya di Firebase:', fbOnly.join(', '));
  }
  
  // Inovasi hanya di localStorage
  const localOnly = Object.keys(localData).filter(k => !fbData[k.replace(/\s+/g, '_')]);
  if (localOnly.length > 0) {
    console.log('\n💾 Hanya di localStorage:', localOnly.join(', '));
  }
  
  // Check sync status
  if (fbOnly.length === 0 && localOnly.length === 0) {
    console.log('\n✅ SYNC STATUS: Perfect sync!');
  } else {
    console.log('\n⚠️  SYNC STATUS: Out of sync - run force reload (#3)');
  }
  
  console.log('═'.repeat(60));
});
